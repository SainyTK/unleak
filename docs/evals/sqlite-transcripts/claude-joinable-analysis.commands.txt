node .claude/skills/unleak/scripts/query.mjs --connection retail_ops --dry-run --sql "
SELECT
  c.id AS customer_id,
  a.account_id,
  c.status AS customer_status,
  c.city,
  c.country,
  c.signup_date,
  c.vip_score,
  a.plan_type,
  a.monthly_revenue AS account_monthly_revenue,
  a.risk_score,
  a.health_status,
  o.currency,
  COUNT(*) AS order_count,
  SUM(o.amount) AS total_spend
FROM orders o
JOIN customers c ON o.customer_id = c.id
JOIN accounts a ON o.account_id = a.account_id
GROUP BY c.id, a.account_id, c.status, c.city, c.country, c.signup_date, c.vip_score, a.plan_type, a.monthly_revenue, a.risk_score, a.health_status, o.currency
ORDER BY total_spend DESC
LIMIT 20
"
