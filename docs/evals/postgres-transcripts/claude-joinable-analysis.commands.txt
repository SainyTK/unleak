node .claude/skills/unleak/scripts/query.mjs --connection retail_ops_pg --dry-run --sql "
SELECT
  o.customer_id AS customer_token,
  o.account_id  AS account_token,
  c.status       AS customer_status,
  c.city,
  c.country,
  c.signup_date,
  c.vip_score,
  a.plan_type,
  a.monthly_revenue,
  a.risk_score,
  a.health_status,
  o.currency,
  COUNT(o.order_id)  AS order_count,
  SUM(o.amount)      AS total_spent
FROM orders o
JOIN customers c ON o.customer_id = c.id
JOIN accounts  a ON o.account_id  = a.account_id
GROUP BY
  o.customer_id, o.account_id,
  c.status, c.city, c.country, c.signup_date, c.vip_score,
  a.plan_type, a.monthly_revenue, a.risk_score, a.health_status,
  o.currency
ORDER BY total_spent DESC
LIMIT 20
"
